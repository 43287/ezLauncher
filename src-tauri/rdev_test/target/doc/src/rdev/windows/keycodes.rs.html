5macro_rules! decl_keycodes {
6    ($($key:ident, $code:literal),*) => {
7        //TODO: make const when rust lang issue #49146 is fixed
8        pub fn code_from_key(key: Key) -> Option<WORD> {
9            match key {
10                $(
11                    Key::$key => Some($code),
12                )*
13                Key::Unknown(code) => Some(code.try_into().ok()?),
14                _ => None,
15            }
16        }
17
18        //TODO: make const when rust lang issue #49146 is fixed
19        pub fn key_from_code(code: WORD) -> Key {
20            match code {
21                $(
22                    $code => Key::$key,
23                )*
24                _ => Key::Unknown(code.into())
25            }
26        }
27    };
28}
